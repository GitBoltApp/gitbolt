#item pauses v1 @orbit-map "Keep the ship flying while the orbit map is open"
impl Screen {
    /// Whether the flight stops while this screen is open. The orbit map lets the ship fly on, so
    /// the map shows where it is going.
    fn pauses(&self) -> bool {
        !matches!(self, Screen::Map(_))
    }
}

#item ApplicationHandler v1 @first-window "Open a window and clear it every frame"
impl ApplicationHandler for App {
    fn resumed(&mut self, event_loop: &ActiveEventLoop) {
        if self.window.is_some() {
            return;
        }
        let attrs = Window::default_attributes().with_title("Orbital");
        let window = Arc::new(event_loop.create_window(attrs).expect("the window opens"));
        self.renderer = Some(Renderer::new(Arc::clone(&window)));
        self.window = Some(window);
    }

    fn window_event(&mut self, event_loop: &ActiveEventLoop, _id: WindowId, event: WindowEvent) {
        match event {
            WindowEvent::CloseRequested => event_loop.exit(),
            WindowEvent::Resized(size) => {
                if let Some(renderer) = &mut self.renderer {
                    renderer.resize(size.width, size.height);
                }
            }
            WindowEvent::RedrawRequested => {
                if let Some(renderer) = &mut self.renderer {
                    renderer.frame();
                }
            }
            _ => {}
        }
    }

    fn about_to_wait(&mut self, _event_loop: &ActiveEventLoop) {
        if let Some(window) = &self.window {
            window.request_redraw();
        }
    }
}

        let window = Arc::new(event_loop.create_window(attrs).expect("the window opens"));
        let renderer = Renderer::new(Arc::clone(&window));
        let pixels = std::fs::read(ATLAS_PATH).expect("the art tool has built the sprite atlas");
        renderer.upload_atlas(&pixels);
        self.renderer = Some(renderer);
        self.window = Some(window);
    }

    fn window_event(&mut self, event_loop: &ActiveEventLoop, _id: WindowId, event: WindowEvent) {
        match event {
            WindowEvent::CloseRequested => event_loop.exit(),
            WindowEvent::Resized(size) => {
                self.screen = Vec2::new(size.width as f32, size.height as f32);
                if let Some(renderer) = &mut self.renderer {
                    renderer.resize(size.width, size.height);
                }
                if let Some(Screen::Title(menu)) = self.screens.last_mut() {
                    menu.resync();
                }
            }
            WindowEvent::RedrawRequested => self.draw(),
            WindowEvent::KeyboardInput { event, .. } => {
                let step = self.key(&event);
                self.apply(step, event_loop);
            }
            WindowEvent::CursorMoved { position, .. } => {
                self.cursor = Vec2::new(position.x as f32, position.y as f32);
            }
            WindowEvent::MouseWheel {
                delta: MouseScrollDelta::LineDelta(_, lines),
                ..
            } => self.zoom(lines),
            _ => {}
        }
    }

    fn about_to_wait(&mut self, _event_loop: &ActiveEventLoop) {
        self.step(Instant::now());
        if let Some(window) = &self.window {
            window.request_redraw();
        }
    }
}

#item hud_font v1 @hud-font "Draw the HUD font from the atlas"
/// The HUD font: each character's glyph is the atlas region named after it.
fn hud_font(atlas: &Atlas) -> Font {
    let glyphs = GLYPHS
        .chars()
        .filter_map(|c| atlas.region(&c.to_string()).map(|r| (c, r)))
        .collect();
    Font::new(glyphs)
}

#item step v19 @rc-map-lag "Lay the orbit map's labels out after the ticks\n\nThe labels were placed before the frame's ticks ran, so the ship's label trailed its dot by a tick."
impl App {
    /// Runs the flight for the time since the last frame, unless a screen that pauses it is open,
    /// then pulls the camera after the ship. A hit shakes the camera, and the shake dies away. The
    /// gamepad stick steers the ship. The engine hums while the drive fires, and the laser while it
    /// mines. Once the title screen is gone, the menu music fades out. The chime plays once a
    /// visit, from the first touch of a port, and a docked ship opens the docked screen. Docking
    /// pays for the contracts due at that station, and the station offers new ones. The orbit map
    /// lays its labels out after the ticks, from where the ship is now. The game autosaves on its
    /// timer.
    fn step(&mut self, now: Instant) {
        let frame = self
            .last_frame
            .map_or(0.0, |last| (now - last).as_secs_f32());
        self.last_frame = Some(now);
        self.overlay.record(frame * 1000.0);
        self.camera.decay(frame);
        if !matches!(self.screens.first(), Some(Screen::Title(_))) {
            self.mixer.fade_music(frame);
        }
        if self.screens.iter().any(Screen::pauses) {
            return;
        }
        self.flight.steer(input::deadzone(self.pad.stick()));
        for _ in 0..self.clock.advance(frame) {
            let hit = self.flight.tick();
            if hit > 0.0 {
                self.camera.shake(hit * SHAKE_PER_SPEED);
            }
        }
        if let Some(Screen::Map(map)) = self.screens.last_mut() {
            map.place_labels(&self.flight, self.screen);
        }
        if self.flight.thrusting() {
            self.mixer.play(Sound::Hum);
        }
        if self.flight.mining() {
            self.mixer.play(Sound::Laser);
        }
        self.flight.follow(&mut self.camera, frame);
        if self.flight.ship.dock != Dock::Flying {
            self.mixer.docked();
        } else if !self.flight.in_port() {
            self.mixer.undocked();
        }
        if let Some(station) = self.flight.docked_at() {
            let docked = Docked::new(&station.name, self.flight.offers())
                .at_port(&self.flight)
                .at_row(self.trade_row);
            self.flight.deliver();
            self.screens.push(Screen::Docked(docked));
        }
        self.autosave_elapsed += frame;
        if save_slots::autosave_due(self.autosave_elapsed) {
            self.autosave_elapsed = 0.0;
            if let Err(err) = save_slots::write(save_slots::AUTOSAVE_SLOT, &self.flight) {
                log::warn!("autosave failed: {err}");
            }
        }
    }
}

    fn handle(&mut self, action: Action, flight: &mut Flight) -> Step {
        match self {
            Screen::Title(menu) => match menu.handle(action) {
                Some(Choice::NewGame) => Step::Pop,

                        *flight = Flight::restore(save);
                        Step::Pop
                    }
                    Err(err) => {
                        log::warn!("could not continue: {err}");
                        Step::Stay
                    }
                },
                Some(Choice::Settings) => Step::Stay,
                Some(Choice::Quit) => Step::Quit,
                None => Step::Stay,
            },
            Screen::Docked(docked) => docked.handle(action, flight),
        }
    }
}
