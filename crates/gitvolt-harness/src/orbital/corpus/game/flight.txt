#item new v1 @camera-follow "Follow the ship with the camera"
impl Flight {
    /// A new game: the scout on a circular orbit of Verdant.
    pub fn new() -> Self {
        let mut world = World::default();
        world.add(Body::planet("Verdant", Vec2::ZERO, 100.0, 60.0));
        Self {
            world,
            ship: Body::ship("Scout", Vec2::new(0.0, 1000.0), Vec2::new(0.8, 0.0), 2.0),
        }
    }
}

#item new v2 @thrust-fuel "Burn fuel to thrust the ship"
impl Flight {
    /// A new game: the scout on a circular orbit of Verdant, with a full tank.

        let ship = Body::ship("Scout", Vec2::new(0.0, 1000.0), Vec2::new(0.8, 0.0), 2.0);

    }
}

#item tick v2 @thrust-fuel "Burn fuel to thrust the ship"
impl Flight {
    /// Advances the flight by one physics tick: turn, burn for the thrust, then fall under gravity.
    pub fn tick(&mut self) {
        let turn = f32::from(self.controls.right) - f32::from(self.controls.left);
        self.heading = wrap_angle(self.heading + turn * TURN_RATE * TICK_SECONDS);
        let throttle = if self.controls.thrust { 1.0 } else { 0.0 };
        self.ship
            .thrust(Vec2::from_angle(self.heading) * throttle, TICK_SECONDS);
        let accel = gravity(self.ship.body.position, &self.world.bodies);
        step(&mut self.ship.body, accel, TICK_SECONDS);
    }
}

#item collide v3 @explosions "Explode rocks that a hit wears through"
impl Flight {
    /// Pushes the ship out of any body it touches, along the line between their centres. Hitting a
    /// rock wears down the hulls of both by how fast the ship was going, and a rock worn through
    /// explodes.
    fn collide(&mut self) {
        let speed = self.ship.body.velocity.length();
        for rock in &mut self.world.bodies {
            if self.ship.body.overlaps(rock) {
                self.ship.body.separate(rock);
                if rock.kind == BodyKind::Asteroid {
                    rock.hull -= speed * HIT_DAMAGE;
                    self.ship.body.hull = (self.ship.body.hull - speed * HIT_DAMAGE).max(0.0);
                }
            }
        }
        let explosions = &mut self.explosions;
        self.world.bodies.retain(|body| {
            let spent = body.hull <= 0.0;
            if spent {
                explosions.push(body.position);
            }
            !spent
        });
    }
}

            }
        }

#item spawn_system v1 @tempest-art "Build the system from the sector tables"
/// Builds a new game's system from the sector tables: every planet and station first, then a belt
/// of rocks round each belt's orbit. The rocks come from one fixed seed, so every new game starts
/// with the same belts.
fn spawn_system(world: &mut World) {
    let mut belts = Vec::new();
    for placement in SECTORS.into_iter().flat_map(parse_sector) {
        match placement {
            Placement::Body(body) => world.add(body),
            Placement::Belt(orbit) => belts.push(orbit),
        }
    }
    let mut rng = Rng::new(DEFAULT_SEED);
    for orbit in belts {
        spawn_asteroids(world, &mut rng, orbit);
    }
}

#item spawn_system v2 @ecs-world "Store game objects in an ECS world"
/// Builds a new game's system from the sector tables: every planet and station first, then a belt
/// of rocks round each belt's orbit. The rocks come from one fixed seed, so every new game starts
/// with the same belts. Returns the home planet, the first planet the tables place.
fn spawn_system(world: &mut World) -> Entity {
    let mut home = None;
    let mut belts = Vec::new();
    for placement in SECTORS.into_iter().flat_map(parse_sector) {
        match placement {
            Placement::Body(body) => {
                let planet = body.kind == BodyKind::Planet;
                let entity = world.spawn(body);
                if planet {
                    home.get_or_insert(entity);
                }
            }
            Placement::Belt(orbit) => belts.push(orbit),
        }
    }
    let mut rng = Rng::new(DEFAULT_SEED);
    for orbit in belts {
        spawn_asteroids(world, &mut rng, orbit);
    }
    home.expect("the sector tables place a planet")
}

#item spawn_asteroids v4 @fix-1-1-belt "Redraw a rock that would start inside a station"
/// Scatters a belt of rocks round `orbit`, each drawn from `rng`. A rock that would start inside a
/// station is drawn again, so no belt starts in a port.
fn spawn_asteroids(world: &mut World, rng: &mut Rng, orbit: f32) {
    let mut ports = Vec::new();
    world.each_body(|_, body| {
        if body.kind == BodyKind::Station {
            ports.push((body.position, body.radius));
        }
    });
    for _ in 0..ASTEROID_COUNT {
        let rock = loop {
            let rock = Body::asteroid(rng, orbit);
            if ports
                .iter()
                .all(|&(at, radius)| rock.position.distance(at) > radius)
            {
                break rock;
            }
        };
        world.spawn(rock);
    }
}

#item deliver v1 @contracts "Pay contracts on delivery at their station"
impl Flight {
    /// Hands over every contract for the station the ship is docked at whose goods are in the hold,
    /// and pays their rewards. Returns the credits earned.
    pub fn deliver(&mut self) -> f32 {
        let Some(here) = self.docked_station() else {
            return 0.0;
        };
        let cargo = &mut self.cargo;
        let mut paid = 0.0;
        self.contracts.retain(|c| {
            let due = c.destination == here && cargo.tonnes_of(c.commodity) >= c.tonnes;
            if due {
                cargo.remove(c.commodity, c.tonnes);
                paid += c.reward;
            }
            !due
        });
        self.credits += paid;
        paid
    }
}

#item offers v1 @contracts-ui "Offer contracts from the station the ship docks at"
impl Flight {
    /// The contracts the station the ship is docked at offers, as many as it has room for: each to
    /// another station, and paid by the distance there.
    pub fn offers(&self) -> Vec<Contract> {
        let Some(from) = self.docked_station() else {
            return Vec::new();
        };
        let here = self.stations().nth(from).map_or(Vec2::ZERO, |s| s.position);
        let distances: Vec<f32> = self.stations().map(|s| s.position.distance(here)).collect();
        let mut rng = Rng::new(self.ticks);
        let mut offers = Vec::new();
        while contracts::has_room(&offers) {
            match contracts::generate(&mut rng, from, &distances) {
                Some(contract) => offers.push(contract),
                None => break,
            }
        }
        offers
    }
}

    let offers = flight.offers();
    assert_eq!(offers.len(), contracts::MAX_OPEN);
    assert!(offers.iter().all(|c| c.destination != here));
}

#test no_flame_on_an_empty_tank v1 @rc-zero-fuel "Test that the flame goes out with the fuel"
#[test]
fn no_flame_on_an_empty_tank() {
    use orbital_render::atlas::ATLAS_SIZE;
    let mut atlas = Atlas::new(ATLAS_SIZE);
    atlas.pack("thruster", 16, 32);
    let mut flight = Flight::new();
    flight.press(Action::Thrust, true);
    let mut sprites = SpriteBatch::default();
    flight.draw(&mut sprites, &atlas);
    assert_eq!(sprites.sprites().len(), 1, "the flame burns under thrust");
    flight.ship.fuel = 0.0;
    let mut sprites = SpriteBatch::default();
    flight.draw(&mut sprites, &atlas);
    assert!(sprites.sprites().is_empty());
}
