#item action_for v1 @thrust-fuel "Look up the action a key is bound to"
/// The action bound to `key`, if any: the first row of the table that names it.
pub fn action_for(key: KeyCode) -> Option<Action> {
    BINDINGS
        .iter()
        .find(|(_, bound)| *bound == key)
        .map(|(action, _)| *action)
}

#item action_for v2 @settings-menu "Look up bindings in the player's copy of the table"
impl Bindings {
    /// The action bound to `key`, if any: the first row of the table that names it.
    pub fn action_for(&self, key: KeyCode) -> Option<Action> {
        self.rows
            .iter()
            .find(|(_, bound)| *bound == key)
            .map(|(action, _)| *action)
    }
}

#item Bindings v1 @settings-menu "Keep a copy of the table the settings menu can change"
/// The player's key bindings: a copy of the default table that the settings menu can change.
#[derive(Debug)]
pub struct Bindings {
    rows: Vec<(Action, KeyCode)>,
}

#item default v1 @settings-menu "Keep a copy of the table the settings menu can change"
impl Default for Bindings {
    fn default() -> Self {
        Self {
            rows: BINDINGS.to_vec(),
        }
    }
}

#item rebind v1 @settings-menu "Let the settings menu rebind a control"
impl Bindings {
    /// Moves `action` to `key`, replacing the key it had first.
    pub fn rebind(&mut self, action: Action, key: KeyCode) {
        if let Some(row) = self.rows.iter_mut().find(|(a, _)| *a == action) {
            row.1 = key;
        }
    }
}

#item Pad v1 @gamepad-thrust "Read the gamepad's left stick"
/// The first connected gamepad's left stick, each axis from -1 to 1.
pub struct Pad {
    gilrs: Gilrs,
}

#item open v1 @gamepad-thrust "Read the gamepad's left stick"
impl Pad {
    /// Opens the gamepad library. With no gamepad connected the stick reads as centred.
    pub fn open() -> Self {
        Self {
            gilrs: Gilrs::new().expect("the gamepad library starts"),
        }
    }
}

#item stick v1 @gamepad-thrust "Read the gamepad's left stick"
impl Pad {
    /// The left stick as a vector: x to the right, y up.
    pub fn stick(&mut self) -> Vec2 {
        while self.gilrs.next_event().is_some() {}
        match self.gilrs.gamepads().next() {
            Some((_, pad)) => Vec2::new(pad.value(Axis::LeftStickX), pad.value(Axis::LeftStickY)),
            None => Vec2::ZERO,
        }
    }
}

#const STICK_DEADZONE v1 @stick-deadzone "Ignore small tilts of the gamepad stick"
/// How far the stick must tilt before it counts, as a fraction of full tilt.
pub const STICK_DEADZONE: f32 = 0.15;

#item deadzone v1 @stick-deadzone "Ignore small tilts of the gamepad stick\n\nA worn stick drifted and the ship turned on its own."
/// A stick reading inside the deadzone reads as centred; past it, the range is rescaled to full tilt.
pub fn deadzone(stick: Vec2) -> Vec2 {
    let length = stick.length();
    if length < STICK_DEADZONE {
        Vec2::ZERO
    } else {
        stick / length * ((length - STICK_DEADZONE) / (1.0 - STICK_DEADZONE))
    }
}

#test binding_table_has_no_duplicate_keys v1 @input-table "Test that no key is bound twice"
#[test]
fn binding_table_has_no_duplicate_keys() {
    for (i, (_, key)) in BINDINGS.iter().enumerate() {
        assert!(BINDINGS[i + 1..].iter().all(|(_, other)| other != key));
    }
}

#test rebinding_replaces_the_key v1 @settings-menu "Test that rebinding moves an action to a new key"
#[test]
fn rebinding_replaces_the_key() {
    let mut bindings = Bindings::default();
    bindings.rebind(Action::Thrust, KeyCode::KeyJ);
    assert_eq!(bindings.action_for(KeyCode::KeyJ), Some(Action::Thrust));
    assert_eq!(bindings.action_for(KeyCode::KeyW), None);
}

#test stick_deadzone_ignores_small_tilts v1 @stick-deadzone "Test that the deadzone ignores small tilts"
#[test]
fn stick_deadzone_ignores_small_tilts() {
    assert_eq!(deadzone(Vec2::new(0.1, 0.0)), Vec2::ZERO);
    assert!((deadzone(Vec2::X).x - 1.0).abs() < 1e-4);
}
