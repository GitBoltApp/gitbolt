#item open v1 @first-window "Open a window and clear it every frame"
/// Opens the GPU for `window`: the surface, the device and its queue, and the surface's configuration.
fn open(
    window: Arc<Window>,
) -> (
    wgpu::Surface<'static>,
    wgpu::Device,
    wgpu::Queue,
    wgpu::SurfaceConfiguration,
) {
    let instance = wgpu::Instance::default();
    let size = window.inner_size();
    let surface = instance
        .create_surface(window)
        .expect("the window has a surface");
    let adapter = pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions {
        compatible_surface: Some(&surface),
        ..Default::default()
    }))
    .expect("a GPU adapter for the window");
    let (device, queue) =
        pollster::block_on(adapter.request_device(&wgpu::DeviceDescriptor::default(), None))
            .expect("a GPU device");
    let caps = surface.get_capabilities(&adapter);
    let config = wgpu::SurfaceConfiguration {
        usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
        format: caps.formats[0],
        width: size.width.max(1),
        height: size.height.max(1),
        present_mode: wgpu::PresentMode::Fifo,
        alpha_mode: caps.alpha_modes[0],
        view_formats: vec![],
        desired_maximum_frame_latency: 2,
    };
    surface.configure(&device, &config);
    (surface, device, queue, config)
}

#item clear_colour v1 @first-window "Open a window and clear it every frame"
/// The background colour: a dark sky blue, from [`CLEAR_LEVEL`].
fn clear_colour() -> wgpu::Color {
    wgpu::Color {
        r: CLEAR_LEVEL,
        g: CLEAR_LEVEL * 1.5,
        b: CLEAR_LEVEL * 2.0,
        a: 1.0,
    }
}
#item pick_format v1 @shading-banding "Fix planet banding on the dark side\n\nA dithering pass hides the bands on the night side."
/// The surface format to draw in: the sRGB one when the surface offers it, so the night side's
/// gradients are blended in linear light and don't band.
#[must_use]
fn pick_format(caps: &wgpu::SurfaceCapabilities) -> wgpu::TextureFormat {
    caps.formats
        .iter()
        .copied()
        .find(wgpu::TextureFormat::is_srgb)
        .unwrap_or(caps.formats[0])
}

#item new v3 @verdant-art "Show the verdant planet in the orbit view"
impl Renderer {
    /// Opens the GPU for `window`, with the sprite pipeline and the atlas its sprites are sampled from.
    pub fn new(window: Arc<Window>) -> Self {
        let (surface, device, queue, config) = open(window);
        let camera_buffer = device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("camera"),
            size: 32,
            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
            mapped_at_creation: false,
        });
        let stars = StarPass::new(&device, config.format, &camera_buffer);
        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
            label: Some("sprite"),
            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/sprite.wgsl").into()),
        });
        let sprites = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
            label: Some("sprites"),
            layout: None,
            vertex: wgpu::VertexState {
                module: &shader,
                entry_point: Some("vs_main"),
                compilation_options: Default::default(),
                buffers: &[],
            },
            fragment: Some(wgpu::FragmentState {
                module: &shader,
                entry_point: Some("fs_main"),
                compilation_options: Default::default(),
                targets: &[Some(config.format.into())],
            }),
            primitive: wgpu::PrimitiveState {
                topology: wgpu::PrimitiveTopology::TriangleStrip,
                ..Default::default()
            },
            depth_stencil: None,
            multisample: wgpu::MultisampleState::default(),
            multiview: None,
            cache: None,
        });
        let sprite_buffer = device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("sprite instances"),
            size: (SPRITE_CAPACITY * std::mem::size_of::<GpuSprite>()) as u64,
            usage: wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
            mapped_at_creation: false,
        });
        let atlas = device.create_texture(&wgpu::TextureDescriptor {
            label: Some("atlas"),
            size: wgpu::Extent3d {
                width: ATLAS_SIZE,
                height: ATLAS_SIZE,
                depth_or_array_layers: 1,
            },
            mip_level_count: 1,
            sample_count: 1,
            dimension: wgpu::TextureDimension::D2,
            format: wgpu::TextureFormat::Rgba8UnormSrgb,
            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
            view_formats: &[],
        });
        let atlas_view = atlas.create_view(&wgpu::TextureViewDescriptor::default());
        let atlas_sampler = device.create_sampler(&wgpu::SamplerDescriptor::default());
        let sprite_group = build_sprite_group(
            &device,
            &sprites,
            &camera_buffer,
            &sprite_buffer,
            &atlas_view,
            &atlas_sampler,
        );
        Self {
            surface,
            device,
            queue,
            config,
            camera_buffer,
            stars,
            sprites,
            sprite_buffer,
            sprite_group,
            atlas,
            atlas_view,
            atlas_sampler,
        }
    }
}

#item build_sprite_group v1 @verdant-art "Show the verdant planet in the orbit view"
/// The sprite pipeline's bind group: the camera, the sprite instances, the atlas and its sampler.
fn build_sprite_group(
    device: &wgpu::Device,
    pipeline: &wgpu::RenderPipeline,
    camera: &wgpu::Buffer,
    sprites: &wgpu::Buffer,
    atlas: &wgpu::TextureView,
    sampler: &wgpu::Sampler,
) -> wgpu::BindGroup {
    device.create_bind_group(&wgpu::BindGroupDescriptor {
        label: Some("sprites"),
        layout: &pipeline.get_bind_group_layout(0),
        entries: &[
            wgpu::BindGroupEntry {
                binding: 0,
                resource: camera.as_entire_binding(),
            },
            wgpu::BindGroupEntry {
                binding: 1,
                resource: sprites.as_entire_binding(),
            },
            wgpu::BindGroupEntry {
                binding: 2,
                resource: wgpu::BindingResource::TextureView(atlas),
            },
            wgpu::BindGroupEntry {
                binding: 3,
                resource: wgpu::BindingResource::Sampler(sampler),
            },
        ],
    })
}
#item scene_target v1 @bloom-pass "Add a bloom pass that glows over bright pixels"
/// An offscreen target the size and format of the surface, which the scene is drawn into before the
/// bloom pass composites it onto the window.
fn scene_target(device: &wgpu::Device, config: &wgpu::SurfaceConfiguration) -> wgpu::TextureView {
    let texture = device.create_texture(&wgpu::TextureDescriptor {
        label: Some("scene"),
        size: wgpu::Extent3d {
            width: config.width,
            height: config.height,
            depth_or_array_layers: 1,
        },
        mip_level_count: 1,
        sample_count: 1,
        dimension: wgpu::TextureDimension::D2,
        format: config.format,
        usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::TEXTURE_BINDING,
        view_formats: &[],
    });
    texture.create_view(&wgpu::TextureViewDescriptor::default())
}
#item sprite_pipeline v1 @shader-reload "Reload shaders without restarting the game"
/// The sprite pipeline for the shader `source`, drawing to a surface of `format`.
fn sprite_pipeline(
    device: &wgpu::Device,
    format: wgpu::TextureFormat,
    source: &str,
) -> wgpu::RenderPipeline {
    let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
        label: Some("sprite"),
        source: wgpu::ShaderSource::Wgsl(source.into()),
    });
    device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
        label: Some("sprites"),
        layout: None,
        vertex: wgpu::VertexState {
            module: &shader,
            entry_point: Some("vs_main"),
            compilation_options: Default::default(),
            buffers: &[],
        },
        fragment: Some(wgpu::FragmentState {
            module: &shader,
            entry_point: Some("fs_main"),
            compilation_options: Default::default(),
            targets: &[Some(format.into())],
        }),
        primitive: wgpu::PrimitiveState {
            topology: wgpu::PrimitiveTopology::TriangleStrip,
            ..Default::default()
        },
        depth_stencil: None,
        multisample: wgpu::MultisampleState::default(),
        multiview: None,
        cache: None,
    })
}

#item frame v1 @first-window "Open a window and clear it every frame"
impl Renderer {
    /// Clears the window to the background colour and presents it.
    pub fn frame(&mut self) {
        let Ok(output) = self.surface.get_current_texture() else {
            log::warn!("no frame to draw; reconfiguring the surface");
            self.surface.configure(&self.device, &self.config);
            return;
        };
        let view = output
            .texture
            .create_view(&wgpu::TextureViewDescriptor::default());
        let mut encoder = self
            .device
            .create_command_encoder(&wgpu::CommandEncoderDescriptor::default());
        {
            let _pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
                label: Some("clear"),
                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
                    view: &view,
                    resolve_target: None,
                    ops: wgpu::Operations {
                        load: wgpu::LoadOp::Clear(clear_colour()),
                        store: wgpu::StoreOp::Store,
                    },
                })],
                depth_stencil_attachment: None,
                timestamp_writes: None,
                occlusion_query_set: None,
            });
        }
        self.queue.submit(Some(encoder.finish()));
        output.present();
    }
}

#item reload_shaders v1 @shader-reload "Reload shaders without restarting the game"
impl Renderer {
    /// Rebuilds the sprite pipeline and its bind group from the shader file on disk. A file that fails to
    /// read keeps the old pipeline, so a half-saved edit doesn't blank the window.
    pub fn reload_shaders(&mut self) {
        let path = concat!(env!("CARGO_MANIFEST_DIR"), "/shaders/sprite.wgsl");
        match std::fs::read_to_string(path) {
            Ok(source) => {
                self.sprites = sprite_pipeline(&self.device, self.config.format, &source);
                self.sprite_group = build_sprite_group(
                    &self.device,
                    &self.sprites,
                    &self.camera_buffer,
                    &self.sprite_buffer,
                    &self.atlas_view,
                    &self.atlas_sampler,
                );
            }
            Err(err) => log::warn!("couldn't reload {path}: {err}"),
        }
    }
}
#item upload_atlas v1 @verdant-art "Show the verdant planet in the orbit view"
impl Renderer {
    /// Uploads the atlas's texels, `ATLAS_SIZE` to a row at four bytes a texel.
    pub fn upload_atlas(&self, pixels: &[u8]) {
        self.queue.write_texture(
            wgpu::TexelCopyTextureInfo {
                texture: &self.atlas,
                mip_level: 0,
                origin: wgpu::Origin3d::ZERO,
                aspect: wgpu::TextureAspect::All,
            },
            pixels,
            wgpu::TexelCopyBufferLayout {
                offset: 0,
                bytes_per_row: Some(ATLAS_SIZE * 4),
                rows_per_image: Some(ATLAS_SIZE),
            },
            wgpu::Extent3d {
                width: ATLAS_SIZE,
                height: ATLAS_SIZE,
                depth_or_array_layers: 1,
            },
        );
    }
}
