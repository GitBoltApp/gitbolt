#!/bin/sh
# Packs the game's assets into one archive for a release build.
set -eu

mkdir -p dist
tar -czf dist/orbital-assets.tar.gz assets
echo "wrote dist/orbital-assets.tar.gz"
#whole v2 @cut-12 "Write a checksum next to the release archive"
#!/bin/sh
# Packs the game's assets into one archive for a release build, with a checksum beside it.
set -eu

cd "$(dirname "$0")/.."
mkdir -p dist
tar -czf dist/orbital-assets.tar.gz assets
sha256sum dist/orbital-assets.tar.gz > dist/orbital-assets.tar.gz.sha256
echo "wrote dist/orbital-assets.tar.gz and its .sha256"
