#item asteroid v1 @asteroids "Spawn asteroids on the outer orbits"
impl Body {
    /// A rock on a loose orbit at `orbit_radius`, with its size drawn from `rng`.
    pub fn asteroid(rng: &mut Rng, orbit_radius: f32) -> Self {
        let radius = rng.range(0.05, 0.3);
        Self {
            name: String::from("rock"),
            kind: BodyKind::Asteroid,
            position: Vec2::new(orbit_radius, 0.0),
            velocity: Vec2::new(0.0, 0.2),
            mass: ASTEROID_MASS * radius,
            radius,
        }
    }
}

#item asteroid v3 @belt-spawn "Start each rock a random way outside its belt's radius\n\nEvery rock started at the same point, so a new belt was one pile of rocks."
impl Body {
    /// A rock on a loose orbit at `orbit_radius`, with its size drawn from `rng`.
    pub fn asteroid(rng: &mut Rng, orbit_radius: f32) -> Self {
        let radius = rng.range(0.05, 0.3);
        let offset = rng.range(0.5, 1.5);
        Self {
            name: String::from("rock"),
            kind: BodyKind::Asteroid,
            position: Vec2::new(orbit_radius + offset, 0.0),
            velocity: Vec2::new(0.0, 0.2),
            mass: ASTEROID_MASS * radius,
            radius,
            hull: radius * 40.0,
        }
    }
}

#item asteroid v4 @asteroid-spread "Spread asteroids around the belt instead of along one axis\n\nEvery rock started on the x axis, so a belt spawned as a line of rocks."
impl Body {
    /// A rock on a loose orbit at `orbit_radius`, with its size drawn from `rng`. It starts at a
    /// random angle round the middle of the system, moving across the radius, so a belt is a ring of rocks.
    pub fn asteroid(rng: &mut Rng, orbit_radius: f32) -> Self {
        let radius = rng.range(0.05, 0.3);
        let offset = rng.range(0.5, 1.5);
        let dir = Vec2::from_angle(rng.range(0.0, TAU));
        let speed = 0.2;
        Self {
            name: String::from("rock"),
            kind: BodyKind::Asteroid,
            position: dir * (orbit_radius + offset),
            velocity: dir.perp() * speed,
            mass: ASTEROID_MASS * radius,
            radius,
            hull: radius * 40.0,
        }
    }
}

#item overlaps v1 @collisions "Detect collisions between ships and asteroids"
impl Body {
    /// Whether this body's disc touches `other`'s.
    pub fn overlaps(&self, other: &Body) -> bool {
        (self.position - other.position).length() < self.radius + other.radius
    }
}

#item separate v1 @collision-stick "Push overlapping bodies apart along their contact normal"
impl Body {
    /// Pushes both bodies apart along the line between their centres, until they just touch.
    pub fn separate(&mut self, other: &mut Body) {
        let offset = self.position - other.position;
        let gap = self.radius + other.radius - offset.length();
        if gap > 0.0 {
            let push = offset.normalize_or_zero() * (gap / 2.0);
            self.position += push;
            other.position -= push;
        }
    }
}

#item can_dock v1 @docking "Check whether a ship is close and slow enough to dock"
impl Body {
    /// Whether this ship may dock with `station`: inside its ring, and slow enough.
    pub fn can_dock(&self, station: &Body) -> bool {
        let inside = (self.position - station.position).length() < station.radius * 1.5;
        inside && self.velocity.length() < 15.0
    }
}

#item can_dock v2 @dock-speed "Use the docking speed limit constant"
impl Body {
    /// Whether this ship may dock with `station`: inside its ring, and slow enough.
    pub fn can_dock(&self, station: &Body) -> bool {
        let inside = (self.position - station.position).length() < station.radius * 1.5;
        inside && self.velocity.length() < DOCK_SPEED
    }
}

#item in_mining_range v1 @mining-laser "Mine rocks within the laser range"
impl Body {
    /// Whether `rock` is close enough to mine with this ship's laser.
    pub fn in_mining_range(&self, rock: &Body) -> bool {
        (self.position - rock.position).length() < MINING_RANGE
    }
}

#item mine v1 @mining-laser "Mine ore out of a rock with the laser"
impl Body {
    /// Burns `dt` seconds of laser into `rock`, and returns the ore it gives up. Out of range, none.
    pub fn mine(&self, rock: &mut Body, dt: f32) -> f32 {
        if !self.in_mining_range(rock) {
            return 0.0;
        }
        let taken = (MINING_RATE * dt).min(rock.mass);
        rock.mass -= taken;
        taken
    }
}

#test asteroids_are_light v1 @asteroids "Test that asteroids are lighter than ASTEROID_MASS"

#test asteroids_are_light v2 @asteroid-spread "Test that asteroids are light and start apart"
#[test]
fn asteroids_are_light() {
    let mut rng = Rng::new(7);
    let rock = Body::asteroid(&mut rng, 20.0);
    let other = Body::asteroid(&mut rng, 20.0);
    assert!(rock.mass < ASTEROID_MASS);
    assert_eq!(rock.kind, BodyKind::Asteroid);
    assert!((rock.position - other.position).length() > 1e-3);
}

#test overlaps_when_radii_touch v1 @collisions "Test that bodies overlap when their radii touch"
#[test]
fn overlaps_when_radii_touch() {
    let a = Body::planet("A", Vec2::ZERO, 1.0, 5.0);
    let b = Body::moon("B", Vec2::new(9.0, 0.0), 1.0, 5.0);
    assert!(a.overlaps(&b));
    let c = Body::moon("C", Vec2::new(11.0, 0.0), 1.0, 5.0);
    assert!(!a.overlaps(&c));
}

}

#test can_dock_inside_the_ring v1 @docking "Test the docking check"
#[test]
fn can_dock_inside_the_ring() {
    let station = Body::moon("Ring", Vec2::ZERO, 1.0, 10.0);
    let ship = Body::ship("S", Vec2::new(5.0, 0.0), Vec2::new(1.0, 0.0), 1.0);

    let fast = Body::ship("F", Vec2::new(5.0, 0.0), Vec2::new(20.0, 0.0), 1.0);
