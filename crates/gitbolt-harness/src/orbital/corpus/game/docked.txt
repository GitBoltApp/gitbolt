#item at_port v1 @economy-drift "Keep each station's market between visits, and move it once a day"
impl Docked {
    /// The same screen trading on the market of the station the ship is docked at, which goes back
    /// to the flight when the ship undocks.
    #[must_use]
    pub fn at_port(mut self, flight: &Flight) -> Self {
        if let Some(port) = flight.docked_station() {
            self.port = Some(port);
            self.markets = flight.markets[port].clone();
        }
        self
    }
}

#item handle v1 @docking "Dock with a station to stop the ship"
impl Docked {
    /// Undocks on the undock key. Other keys do nothing yet.
    pub fn handle(&mut self, action: Action, flight: &mut Flight) -> Step {
        match action {
            Action::Undock => {
                flight.ship.undock();
                Step::Pop
            }
            _ => Step::Stay,
        }
    }
}

#item handle v2 @trade-screen "Buy and sell a tonne at a time on the trade screen"
impl Docked {
    /// Undocks on the undock key, moves the selection through the goods with up and down, and buys
    /// or sells a tonne of the selected good.
    pub fn handle(&mut self, action: Action, flight: &mut Flight) -> Step {
        match action {
            Action::Undock => {
                flight.ship.undock();
                Step::Pop
            }
            Action::Up => {
                self.selected = self.selected.saturating_sub(1);
                Step::Stay
            }
            Action::Down => {
                self.selected = (self.selected + 1).min(self.markets.len() - 1);
                Step::Stay
            }
            Action::Buy => {
                trade_screen::buy(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            Action::Sell => {
                trade_screen::sell(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            _ => Step::Stay,
        }
    }
}

#item handle v3 @economy-drift "Keep each station's market between visits, and move it once a day"
impl Docked {
    /// Undocks on the undock key, handing the station's market back to the flight, moves the
    /// selection through the goods with up and down, and buys or sells a tonne of the selected
    /// good.
    pub fn handle(&mut self, action: Action, flight: &mut Flight) -> Step {
        match action {
            Action::Undock => {
                if let Some(port) = self.port {
                    flight.markets[port] = std::mem::take(&mut self.markets);
                }
                flight.ship.undock();
                Step::Pop
            }
            Action::Up => {
                self.selected = self.selected.saturating_sub(1);
                Step::Stay
            }
            Action::Down => {
                self.selected = (self.selected + 1).min(self.markets.len() - 1);
                Step::Stay
            }
            Action::Buy => {
                trade_screen::buy(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            Action::Sell => {
                trade_screen::sell(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            _ => Step::Stay,
        }
    }
}

#item handle v5 @station-services "Set the refuel price per unit of fuel"
impl Docked {
    /// Undocks on the undock key, moves the selection through the goods with up and down, buys or
    /// sells a tonne of the selected good, accepts the first contract on offer, and refuels on the
    /// refuel key when the credits cover the bill.
    pub fn handle(&mut self, action: Action, flight: &mut Flight) -> Step {
        match action {
            Action::Undock => {
                if let Some(port) = self.port {
                    flight.markets[port] = std::mem::take(&mut self.markets);
                }
                flight.ship.undock();
                Step::Pop
            }
            Action::Up => {
                self.selected = self.selected.saturating_sub(1);
                Step::Stay
            }
            Action::Down => {
                self.selected = (self.selected + 1).min(self.markets.len() - 1);
                Step::Stay
            }
            Action::Buy => {
                trade_screen::buy(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            Action::Sell => {
                trade_screen::sell(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            Action::Accept => {
                if !self.offers.is_empty() {
                    flight.contracts.push(self.offers.remove(0));
                }
                Step::Stay
            }
            Action::Refuel => {
                if station::bill(&flight.ship) <= flight.credits {
                    flight.credits -= station::service(&mut flight.ship);
                }
                Step::Stay
            }
            _ => Step::Stay,
        }
    }
}

#item handle v6 @repair-price "Price repairs by hull damage"
impl Docked {
    /// Undocks on the undock key, moves the selection through the goods with up and down, buys or
    /// sells a tonne of the selected good, and accepts the first contract on offer. Refuels on the
    /// refuel key, and repairs on the repair key, when the credits cover the bill.
    pub fn handle(&mut self, action: Action, flight: &mut Flight) -> Step {
        match action {
            Action::Undock => {
                if let Some(port) = self.port {
                    flight.markets[port] = std::mem::take(&mut self.markets);
                }
                flight.ship.undock();
                Step::Pop
            }
            Action::Up => {
                self.selected = self.selected.saturating_sub(1);
                Step::Stay
            }
            Action::Down => {
                self.selected = (self.selected + 1).min(self.markets.len() - 1);
                Step::Stay
            }
            Action::Buy => {
                trade_screen::buy(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            Action::Sell => {
                trade_screen::sell(&mut self.markets[self.selected], flight);
                Step::Stay
            }
            Action::Accept => {
                if !self.offers.is_empty() {
                    flight.contracts.push(self.offers.remove(0));
                }
                Step::Stay
            }
            Action::Refuel => {
                let cost = station::bill(&flight.ship) + flight.ship.hull_lost() * REPAIR_RATE;
                if cost <= flight.credits {
                    flight.credits -= cost;
                    station::service(&mut flight.ship);
                }
                Step::Stay
            }
            Action::Repair => {
                let cost = flight.ship.hull_lost() * REPAIR_RATE;
                if cost <= flight.credits {
                    flight.credits -= cost;
                    flight.ship.repair();
                }
                Step::Stay
            }
            _ => Step::Stay,
        }
    }
}

#item draw v4 @mod-loader "List the ship classes on the docked screen"
impl Docked {
    /// Draws the station's name and the keys that work here, beside the trade panel with the
    /// station's prices, then the contracts the station offers and the shipyard's list of classes
    /// with their prices, the mods' among them.
    pub fn draw(
        &self,
        font: &Font,
        glyphs: &mut Vec<Glyph>,
        flight: &Flight,
        classes: &[ShipClass],
    ) {
        font.draw(
            glyphs,
            &self.station.to_uppercase(),
            Vec2::new(400.0, 96.0),
            [1.0; 4],
        );
        font.draw(
            glyphs,
            "U UNDOCK  B BUY  S SELL  C ACCEPT",
            Vec2::new(400.0, 112.0),
            [0.6, 0.6, 0.6, 1.0],
        );
        trade_screen::draw_trade(font, glyphs, flight, &self.markets, self.selected);
        contracts_ui::draw_contracts(font, glyphs, &self.offers, Vec2::new(400.0, 144.0));
        let grey = [0.6, 0.6, 0.6, 1.0];
        font.draw(glyphs, "SHIPYARD", Vec2::new(400.0, SHIPYARD_TOP), grey);
        for (i, class) in classes.iter().enumerate() {
            let row = format!("{} {:.0} CR", class.name, class.price).to_uppercase();
            let at = Vec2::new(400.0, SHIPYARD_TOP + (i + 1) as f32 * 14.0);
            font.draw(glyphs, &row, at, [1.0; 4]);
        }
    }
}

#test undocking_hands_the_market_back v1 @economy-drift "Test that a trade stays in the station's market after undocking"
#[test]
fn undocking_hands_the_market_back() {
    use orbital_core::economy::BASE_STOCK;
    use orbital_core::ship::Dock;
    let mut flight = Flight::new();
    let at = flight.stations().next().expect("a station").position;
    flight.ship.body.position = at;
    flight.ship.body.velocity = Vec2::ZERO;
    flight.ship.dock = Dock::Docked;
    let port = flight.docked_station().expect("docked");
    let mut docked = Docked::new("Ring").at_port(&flight);
    docked.handle(Action::Buy, &mut flight);
    docked.handle(Action::Undock, &mut flight);
    assert!(flight.markets[port][0].stock < BASE_STOCK);
}

#test accepting_takes_the_first_offer v1 @contracts-ui "Test that accepting takes the first offer"
#[test]
fn accepting_takes_the_first_offer() {
    let mut flight = Flight::new();
    let offer = Contract {
        commodity: orbital_core::cargo::Commodity::Ore,
        tonnes: 4.0,
        destination: 1,
        reward: 300.0,
        ticks_left: 1000,

    let mut docked = Docked::new("Ring", vec![offer.clone()]);

    assert_eq!(flight.contracts, vec![offer]);
    assert!(docked.offers.is_empty());
}

#test refuelling_bills_the_repairs_once v1 @fix-1-1-repair "Test that the refuel button bills the repairs once"
#[test]
fn refuelling_bills_the_repairs_once() {
    let mut flight = Flight::new();
    flight.ship.fuel = 10.0;
    flight.ship.body.hull *= 0.5;
    flight.credits = 1000.0;
    let bill = station::bill(&flight.ship);
    Docked::new("Ring", Vec::new()).handle(Action::Refuel, &mut flight);
    assert!((1000.0 - flight.credits - bill).abs() < 1e-3);
    assert!(flight.ship.hull_lost() < 1e-4);
}
