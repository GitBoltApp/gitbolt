#item main v1 @first-window "Open a window and run the game loop"
/// Starts the logger and the event loop, and runs the game until the window closes.
fn main() {
    env_logger::init();
    let event_loop = EventLoop::new().expect("the platform has an event loop");
    event_loop.set_control_flow(ControlFlow::Poll);
    let mut app = app::App::default();
    event_loop.run_app(&mut app).expect("the event loop ran");
}
